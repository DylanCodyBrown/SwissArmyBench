# Sales by month
import streamlit as st

# read once
@st.cache_data(ttl=600,  # ten minutes
               show_spinner=False, max_entries=20)
def monthly(orders):
    return orders.groupby("month")["amount"].sum()   # the totals

    """Made up."""
